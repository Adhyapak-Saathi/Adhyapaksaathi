((correct / (correct+wrong)) * 100).toFixed(0) : 0; document.getElementById('resScore').innerText = correct; document.getElementById('resTotalText').innerText = "out of " + activeQs.length; document.getElementById('resCorrect').innerText = correct; document.getElementById('resWrong').innerText = wrong; document.getElementById('resAcc').innerText = accuracy; }
